\section{O problema discreto}

\begin{frame}{Equação diferencial da elastodinâmica}
    \indent Problemas elastodinâmicos são regidos pela equação: $$\begin{aligned}
        \rho \ddot{u} - \nabla \cdot \sigma(u)  = f \hspace{0.5cm} \text{in} \hspace{3mm} \Omega \times (0,T],  \\
        u = g_D \hspace{0.5cm} \text{in} \hspace{3mm} \Gamma_D \times (0,T],\\
        \sigma (u) n = g_N \hspace{0.5cm} \text{in} \hspace{3mm} \Gamma_N \times (0,T],\\
        u = u_0 \hspace{0.5cm} \text{in} \hspace{3mm} \Omega \times \{0\},\\
        \dot{u} = u_1 \hspace{0.5cm} \text{in} \hspace{3mm} \Omega \times \{0\}.
    \end{aligned}$$
\end{frame}

\begin{frame}{Forma variacional }
    O problema variacional  é: $$\begin{cases} \text{find} \hspace{2mm} w(t) \in \mathbb{V} := H^1(\Omega): \\ m(\ddot{w},v) + c(\dot{w},v) + a(w,v) = \tilde{L}(v,t) \hspace{3mm} \forall v \in \mathbb{V} \end{cases},$$ ou,
    $$\int_\Omega \rho \ddot{u} \cdot v \, dx + \int_\Omega \mathbb{D} \dot{u} \cdot v \, dx + \int_\Omega \epsilon(u) \mathbb{C} \epsilon(v) \, dx = \int_\Omega f \cdot v \, dx + \int_{\Gamma_N} g_N \cdot v \, dS$$\pause
\end{frame}

\begin{frame}{O problema de autovalores}
    Em problemas de vibração livre a equação se simplifica para: $$[M] \ddot{u} + [K]u = 0.$$
    As frequências naturais provém da solução do problema de autovalor: $$\left([K]-\omega^2[M]\right)\phi_ = 0,$$
\end{frame}

\section{O Método dos Elementos Virtuais}
\begin{frame}{Espaços de aproximação}
    Para definir o espaço de aproximação do método, primeiro define-se um espaço auxiliar: $$\mathbb{B}_k(\partial E) := \{ v \in C^0(\partial E) : v_{|e} \in \mathbb{P}_k(e) \forall e \subset \partial E \}.$$ \pause
    O espaço de aproximação é dado por: $$V^{\Omega,k}:=\{v \in H^1(\Omega): v_{|\partial\Omega} \in \mathbb{B}_k(\partial\Omega), \nabla^2 v_{\Omega} \in \mathbb{P}_{k-2}(\Omega)\}.$$ \pause 
    O espaço ampliado de \cite{equivalentprojectorsvirtualelementmethods2013a} é:$$\begin{aligned}
        \mathbb{V}^{E,k}:=\{v_h \in H^1(\Omega): v_{|\partial E} \in \mathbb{B}_k(\partial E), \nabla^2 v_{\Omega} \in \mathbb{P}_{k}(E), \\ \int_E (v_h - \Pi_k^\nabla v_h) \mu_h dE \hspace{2mm} \forall\mu_h =0 \in \mathbb{P}_k(E)\backslash \mathbb{P}_{k-2}(E) \}.
    \end{aligned}$$
\end{frame}

\begin{frame}{Projeção}
    As funções de base são desconhecidas, mas o espaço de aproximação é mapeado através de um operador de projeção:$\Pi_k$.
    As funções de base podem ser divididas em dois termos:$$\varphi=\Pi_k\varphi + (I -\Pi_k)\varphi.$$ 
    Nesse trabalho serão trabalhadas três projeções:
    \begin{enumerate}
        \item $\Pi_k^{\nabla,E}: V^{\Omega,k} \rightarrow \left[\mathbb{P}_k(\Omega)\right]^2$
        \item $\Pi_k^{0,E}: V^{\Omega,k} \rightarrow \left[\mathbb{P}_k(\Omega)\right]^2$
        \item $\Pi_{k-1}^{0,E}: V^{\Omega,k} \rightarrow \left[\mathbb{P}_{k-1}(\Omega)\right]^{2 \times 2}$
    \end{enumerate}
\end{frame}

\begin{frame}{Subdivisão da forma bilinear}
    Assim como há a subdivisão das funções de base, também há das formas bilineares. $$\begin{aligned}
        m_{\Omega_v}(\ddot{u_h},v_h) = m_{\Omega_v}(\Pi\ddot{u_h},\Pi v_h) + \tau_h S_{\Omega_v}(\ddot{u_h},v_h) \\
        a(u_h,v_h) = a_{\Omega_v}(\Pi u_h,\Pi v_h) + \tau_h S_{\Omega_v}(u_h,v_h)
    \end{aligned}$$
\end{frame}

\begin{frame}{Elemento virtual}
    \begin{figure}
        \centering
        \includegraphics[width=0.5\linewidth]{figuras/Elemento.png}
    \end{figure} 
    {\centering{\small Fonte: \cite{virtualelementmethodsengineeringsciences2024}}}\\[5mm]
    Os graus de liberdade são:
    \begin{itemize}
        \item Vértices: $n_v$
        \item Arestas: $n_v(k-1)$
        \item internos: $\frac{k(k - 1)}{2}$
    \end{itemize}
\end{frame}


\begin{frame}{Projetores}
    Para o cálculo do projetor, as seguintes equações devem ser satisfeitas $\forall v_h \in \mathbb{V}^{E,k}$: 
    \begin{itemize}
        \item $$\Pi_k^{\nabla,E}: \begin{cases} 
        a^E(\Pi_k v_h,q) = a^E(v_h,q) \hspace{2mm} \forall q\in \mathbb{P}_k(E) \\
        \bar{\Pi_k v_h} = \bar{v_h}, \end{cases}$$ 
        \item $$\Pi_k^{0,E}: (\Pi_k^{\nabla,E}v_h,q) = (v_h,q) \hspace{2mm} \forall q\in \mathbb{P}_k(E) $$ 
        \item $$\Pi_{k-1}^{0,E}: (\Pi_{k-1}^{0,E} \epsilon(v_h),q) = (\epsilon(v_h),q) \hspace{2mm} \forall q\in \mathbb{P}_k(E) $$ 
    \end{itemize} \pause 
\end{frame}

\begin{frame}
    Os operadores de projeção são obtidos solucionando os seguintes sistemas lineares:
    \begin{itemize}
        \item $\mathcal{G}^{\nabla}\Pi_k^{\nabla,E}=\mathcal{B}^{\nabla}$
        \item $\mathcal{G}^0\Pi_{k-1}^{0,E}=\mathcal{B}^0$
        \item $\mathcal{H}\Pi_k^{0,E}=\mathcal{C}$
    \end{itemize}
\end{frame}

\begin{frame}{Matriz $\mathcal{B}$}
    A parcela não é computável. Aplica-se o teorema da divergência:
    $$\begin{aligned} \mathcal{B}^\nabla = \int_{\Gamma_e} (\nabla q \cdot n) v_{h|\Gamma_e} - \int_E \nabla^2(q) v_h \\
                      \mathcal{B}^\epsilon = \int_{\Gamma_e} ( q \cdot n) v_{h|\Gamma_e} - \int_E \nabla \cdot (q) v_h   \end{aligned}$$ \pause
    Parcela de contorno pode ser aproximada por quadratura:
    $$ \begin{aligned} \mathcal{B}^\nabla(v_h,m_i) = \sum_{j=1}^{n_e} l_e\sum_{r=1}^{k+1} w_r \nabla q(\xi_j,\eta_j) n - \{\nabla^2 q(\xi,\eta)\}^T \{m_i\}  \\
                      \mathcal{B}^\nabla(v_h,m_i) = \sum_{j=1}^{n_e} l_e\sum_{r=1}^{k+1} w_r q(\xi_j,\eta_j) n - \{\nabla \cdot q(\xi,\eta)\}^T \{m_i\} \end{aligned}$$
    
\end{frame}

\begin{frame}{Matriz $\mathcal{G}$}
    Obtida por:
    \begin{itemize}
        \item Integração numérica
        \item Integração sobre momentos arbitrários
        \item Para $\Pi_k^{\nabla,E}$ a identidade $\mathcal{G}=\mathcal{B}\mathcal{D}$ é válida.
    \end{itemize}
\end{frame}

\begin{frame}{Matrizes $\mathcal{H}$ e $\mathcal{C}$}
    \begin{itemize}
        \item Matriz $\mathcal{H}$ solucionada igual $\mathcal{G}^0$
        \item Matriz $\mathcal{C}$: $$\mathcal{C}_{\alpha} = \begin{cases}
        (v_h,p_\alpha)_E \hspace{3mm} \text{se} \hspace{2mm} 1 \leq \alpha \leq n_{k-2} \\
        (\Pi_k^\nabla v_h,p_\alpha)_E \hspace{3mm} \text{se} \hspace{2mm} n_{k-2}+1 \leq \alpha \leq n_{k}
    \end{cases}$$
    \end{itemize}
\end{frame}

\begin{frame}{Matriz $\mathcal{D}$}
    \begin{itemize}
        \item Matriz de mudança de bases: polinomial $\rightarrow$ canonica
        \item $\mathcal{D}_{i\alpha} = \chi_i(p_\alpha)$
    \end{itemize}
\end{frame}

\begin{frame}{Métodos de estabilização}
    Diversas técnicas foram propostas na literatura para estabilização:
    \begin{itemize}
        \item \cite{basicprinciplesvirtualelementmethods2013a}: $$S_E = (I - D\Pi_k)^T(I - D\Pi_k);$$
        \item \cite{arbitraryorder2dvirtualelementspolygonalmeshespartelasticproblem2017a}: $$S_E = \left[I - D(D^TD)^{-1}D^T \right];$$
        \item \cite{wriggers2017efficient}: $$U_{stab} = \sum_{v=1}^{n_v}\int_E \bar{\psi}(u_h)dE + \sum_{v=1}^{n_v}\int_E \bar{\psi}(\Pi_k u_h)dE$$
    \end{itemize}    
\end{frame}

\begin{frame}{Parâmetros de estabilização}
    Os parâmetros são diversos na literatura. 
    \begin{itemize}
        \item \cite{numericalrecipeselastodynamicvirtualelementmethodsexplicittimeintegration2020}: $$\begin{aligned}
            \tau_k = \text{max}\left(\frac{tr(K^{c})}{n_{dof}}, \alpha_0 \frac{\mathbb{C}}{3}\right)\\
            \tau_m = \rho |E|     
            \end{aligned}$$
        \item \cite{virtualelementmethodlinearelastodynamicsmodelsconvergencestabilitydissipationdispersionanalysis2019}: $$\begin{aligned}
            \tau_k = \text{max}\left(2\mu , \lambda \right)\\
            \tau_m = \rho h_E^2     
            \end{aligned}$$
    \end{itemize}
\end{frame}
